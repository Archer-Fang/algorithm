class Solution:
    def findRedundantConnection(self, edges: List[List[int]]) -> List[int]:
        result=[]
        visited=[0 for i in range(len(edges)+1)]
        self.dfs(visited,edges,0,result)
        return result

    def dfs(self, visited, edges, index,result):
        if index == len(edges):
            return
        if visited[edges[index][1]]==1:
            result.append(edges[index][0])
            result.append(edges[index][1])
            return
        visited[edges[index][0]]=1
        visited[edges[index][1]]=1
        self.dfs(visited, edges,index+1,result)
