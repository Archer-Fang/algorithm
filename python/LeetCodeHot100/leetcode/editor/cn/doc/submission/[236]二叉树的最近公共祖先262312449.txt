# Definition for a binary tree node.
# class TreeNode:
#     def __init__(self, x):
#         self.val = x
#         self.left = None
#         self.right = None

class Solution:
    def lowestCommonAncestor(self, root: 'TreeNode', p: 'TreeNode', q: 'TreeNode') -> 'TreeNode':

        def dfs(root, p, q):
            if not root:
                return None
            l = dfs(root.left,p,q)
            r = dfs(root.right,p,q)
            if root == q or root == p:
                return root
            if l and r:
                return root
            elif l:
                return l
            else:
                return r
        return dfs(root,p,q)
