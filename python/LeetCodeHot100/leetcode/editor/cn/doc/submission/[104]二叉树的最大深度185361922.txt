/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 */
class TreeNode {
    int val;
    TreeNode left;
    TreeNode right;
    TreeNode() {}
    TreeNode(int val) { this.val = val; }
    TreeNode(int val, TreeNode left, TreeNode right) {
        this.val = val;
        this.left = left;
        this.right = right;
    }
}
class Solution {
    int deep=0;
    public int maxDepth(TreeNode root) {
        dfs(root,0);
        return deep;
    }

    private void dfs(TreeNode root,int d) {
        if(root==null){
            deep=Math.max(deep,d);
            return;
        }
        dfs(root.left,d+1);
        dfs(root.right,d+1);
    }
}
