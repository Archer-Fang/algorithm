class Solution:
    def countSubstrings(self, s: str) -> int:
        n=len(s)
        dp=[[0]*(n) for i in range(n)]
        ans=0
        for i in range(n-1,-1,-1):
            for j in range(i,n):
                if s[i]==s[j]:
                    if abs(i-j)<=1 or dp[i+1][j-1]==1:
                        ans+=1
                        dp[i][j]=1
        return ans

# runtime:248 ms
# memory:22.8 MB
